# ES6+ 常用特性

## 变量声明

### let 和 const

```javascript
// let - 块作用域变量
function testLet() {
  if (true) {
    let x = 10
    console.log(x) // 10
  }
  console.log(x) // ReferenceError
}

// const - 常量
const PI = 3.14159
PI = 3.14 // TypeError

// 对象常量
const person = { name: 'John' }
person.name = 'Jane' // 允许，对象内容可变
person = {} // TypeError，不允许重新赋值
```

### 变量提升对比

```javascript
// var 变量提升
console.log(a) // undefined
var a = 10

// let/const 不会提升
console.log(b) // ReferenceError
let b = 20
```

## 箭头函数

```javascript
// 传统函数
function add(a, b) {
  return a + b
}

// 箭头函数
const add = (a, b) => a + b

// 多行函数体
const calculate = (a, b) => {
  let result = a + b
  return result * 2
}

// 作为回调函数
const numbers = [1, 2, 3, 4, 5]
const doubled = numbers.map(num => num * 2)

// this 绑定
function Timer() {
  this.seconds = 0
  setInterval(() => {
    this.seconds++ // 箭头函数继承外层 this
  }, 1000)
}
```

const html = `

## 解构赋值

### 数组解构

```javascript
const numbers = [1, 2, 3, 4, 5]

// 基本解构
const [first, second] = numbers
console.log(first, second) // 1, 2

// 跳过元素
const [a, , c] = numbers
console.log(a, c) // 1, 3

// 剩余元素
const [head, ...tail] = numbers
console.log(head, tail) // 1, [2, 3, 4, 5]

// 默认值
const [x = 10, y = 20] = [1]
console.log(x, y) // 1, 20
```

### 对象解构

```javascript
const person = {
  name: 'John',
  age: 30,
  job: 'Developer'
}

// 基本解构
const { name, age } = person
console.log(name, age) // John, 30

// 重命名
const { name: fullName, age: years } = person
console.log(fullName, years) // John, 30

// 默认值
const { name, city = 'Unknown' } = person
console.log(city) // Unknown

// 嵌套解构
const user = {
  info: {
    name: 'Jane',
    age: 25
  },
  settings: {
    theme: 'dark'
  }
}

const { info: { name: userName }, settings: { theme } } = user
console.log(userName, theme) // Jane, dark
```

## 展开运算符

### 数组展开

```javascript
const arr1 = [1, 2, 3]
const arr2 = [4, 5, 6]

// 合并数组
const combined = [...arr1, ...arr2] // [1, 2, 3, 4, 5, 6]

// 复制数组
const copy = [...arr1]

// 在函数调用中使用
function sum(a, b, c) {
  return a + b + c
}
sum(...[1, 2, 3]) // 6

// 数组字面量中使用
const arr = [1, ...arr1, 4] // [1, 1, 2, 3, 4]
```

### 对象展开

```javascript
const obj1 = { a: 1, b: 2 }
const obj2 = { c: 3, d: 4 }

// 合并对象
const combined = { ...obj1, ...obj2 } // { a: 1, b: 2, c: 3, d: 4 }

// 覆盖属性
const merged = { ...obj1, b: 20 } // { a: 1, b: 20 }

// 复制对象
const copy = { ...obj1 }
```

## 默认参数和剩余参数

```javascript
// 默认参数
function greet(name = 'World', greeting = 'Hello') {
  return `${greeting}, ${name}!`
}

greet() // Hello, World!
greet('John') // Hello, John!
greet('John', 'Hi') // Hi, John!

// 剩余参数
function sum(...numbers) {
  return numbers.reduce((total, num) => total + num, 0)
}

sum(1, 2, 3) // 6
sum(1, 2, 3, 4, 5) // 15

// 结合默认参数
function multiply(multiplier, ...numbers) {
  return numbers.map(num => num * multiplier)
}

multiply(2, 1, 2, 3) // [2, 4, 6]
```

## Promise 和异步编程

### Promise 基础

```javascript
// 创建 Promise
const promise = new Promise((resolve, reject) => {
  setTimeout(() => {
    const success = Math.random() > 0.5
    if (success) {
      resolve('Success!')
    } else {
      reject(new Error('Failed!'))
    }
  }, 1000)
})

// 使用 Promise
promise
  .then(result => {
    console.log(result) // Success!
  })
  .catch(error => {
    console.error(error) // Error: Failed!
  })
  .finally(() => {
    console.log('Done')
  })

// Promise.all
Promise.all([promise1, promise2, promise3])
  .then(results => {
    console.log(results)
  })

// Promise.race
Promise.race([promise1, promise2])
  .then(result => {
    console.log('First to complete:', result)
  })
```

## 类 (Class)

```javascript
class Person {
  constructor(name, age) {
    this.name = name
    this.age = age
  }

  // 实例方法
  greet() {
    return `Hello, I'm ${this.name}`
  }

  // getter
  get fullInfo() {
    return `${this.name} (${this.age} years old)`
  }

  // setter
  set fullInfo(info) {
    const [name, ageStr] = info.split(' (')
    this.name = name
    this.age = parseInt(ageStr)
  }

  // 静态方法
  static createAnonymous() {
    return new Person('Anonymous', 0)
  }
}

// 继承
class Developer extends Person {
  constructor(name, age, language) {
    super(name, age)
    this.language = language
  }

  code() {
    return `${this.name} is coding in ${this.language}`
  }
}

const dev = new Developer('John', 30, 'JavaScript')
console.log(dev.greet()) // Hello, I'm John
console.log(dev.code())  // John is coding in JavaScript
```

## 模块系统

### 导出和导入

```javascript
// math.js
export const PI = 3.14159

export function add(a, b) {
  return a + b
}

export default function multiply(a, b) {
  return a * b
}

// main.js
import multiply, { PI, add } from './math.js'

console.log(add(1, 2))      // 3
console.log(multiply(3, 4)) // 12
console.log(PI)             // 3.14159

// 重命名导入
import { add as sum } from './math.js'

// 全部导入
import * as math from './math.js'
console.log(math.add(1, 2))
```

## Map 和 Set

### Map

```javascript
const map = new Map()

// 设置值
map.set('name', 'John')
map.set('age', 30)

// 获取值
console.log(map.get('name')) // John

// 检查存在
console.log(map.has('name')) // true

// 删除
map.delete('age')

// 遍历
for (let [key, value] of map) {
  console.log(`${key}: ${value}`)
}

// 获取所有键/值
console.log([...map.keys()])   // ['name']
console.log([...map.values()]) // ['John']
```

### Set

```javascript
const set = new Set([1, 2, 3, 3, 2]) // 自动去重

// 添加元素
set.add(4)
set.add(5)

// 检查存在
console.log(set.has(1)) // true

// 删除
set.delete(2)

// 遍历
for (let item of set) {
  console.log(item)
}

// 集合操作
const setA = new Set([1, 2, 3])
const setB = new Set([2, 3, 4])

// 并集
const union = new Set([...setA, ...setB])

// 交集
const intersection = new Set([...setA].filter(x => setB.has(x)))

// 差集
const difference = new Set([...setA].filter(x => !setB.has(x)))
```

::: tip 最佳实践
- 优先使用 const 和 let 而不是 var
- 使用箭头函数简化回调
- 利用解构赋值简化代码
- 使用 Promise 处理异步操作
- 合理使用模块系统组织代码
:::
